function unique(field){return [...new Set(DATA.map(d=>d[field]))].sort((a,b)=>String(a).localeCompare(String(b),'pt-BR'))}
function fillSelect(id, values){
  const el=document.getElementById(id);
  values.forEach(v=>{const o=document.createElement('option');o.value=v;o.textContent=v;el.appendChild(o)})
}
fillSelect('modelFilter', unique('model'));

['modelFilter','yearFilter','cityFilter','payFilter'].forEach(id=>document.getElementById(id).addEventListener('change',render));

function family(model){
  if(model.startsWith('911')) return '911';
  if(model.startsWith('718')) return '718';
  if(model.startsWith('Taycan')) return 'Taycan';
  if(model.startsWith('Panamera')) return 'Panamera';
  if(model.startsWith('Macan')) return 'Macan';
  if(model.startsWith('Cayenne')) return 'Cayenne';
  return model;
}
function money(v){return v.toLocaleString('pt-BR',{style:'currency',currency:'BRL',maximumFractionDigits:0})}
function pct(v){return (v*100).toFixed(0)+'%'}
function esc(s){return String(s).replace(/[&<>"']/g,m=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[m]))}

function render(){
  const mf=document.getElementById('modelFilter').value;
  const yf=document.getElementById('yearFilter').value;
  const cf=document.getElementById('cityFilter').value;
  const pf=document.getElementById('payFilter').value;
  const rows=DATA.filter(d=>(!mf||d.model===mf)&&(!yf||String(d.year)===yf)&&(!cf||d.city===cf)&&(!pf||d.pay===pf));

  const n=rows.length, revenue=rows.reduce((s,d)=>s+d.price,0), ticket=n?revenue/n:0;
  const years=countBy(rows,d=>d.year);
  const yearEntries=Object.entries(years).sort((a,b)=>b[1]-a[1]);
  const leadYear=yearEntries[0];

  document.getElementById('kpiSales').textContent=n;
  document.getElementById('kpiRevenue').textContent=money(revenue);
  document.getElementById('kpiTicket').textContent=money(ticket);
  document.getElementById('kpiYear').textContent=leadYear?leadYear[0]:'—';
  document.getElementById('kpiYearDetail').textContent=leadYear?`${leadYear[1]} vendas · ${pct(leadYear[1]/n)}`:'Sem dados';

  const families=countBy(rows,d=>family(d.model));
  drawBars('modelChart',Object.entries(families).sort((a,b)=>b[1]-a[1]),n);

  drawBars('yearChart',Object.entries(years).sort((a,b)=>Number(a[0])-Number(b[0])),n);

  const pays=countBy(rows,d=>d.pay);
  const payEntries=Object.entries(pays).sort((a,b)=>b[1]-a[1]);
  drawDonut(payEntries,n);

  drawEvolution(rows);

  const cityMap={};
  rows.forEach(d=>{
    cityMap[d.city]??={state:d.state,models:{}};
    cityMap[d.city].models[d.model]=(cityMap[d.city].models[d.model]||0)+1;
  });
  const cityRows=Object.entries(cityMap).map(([city,v])=>{
    const max=Math.max(...Object.values(v.models));
    const leaders=Object.entries(v.models).filter(x=>x[1]===max).map(x=>x[0]);
    return {city,state:v.state,leaders,max};
  }).sort((a,b)=>a.city.localeCompare(b.city,'pt-BR'));
  document.getElementById('cityTable').innerHTML=cityRows.map(r=>`<tr><td>${esc(r.city)}</td><td>${esc(r.state)}</td><td>${r.leaders.map(x=>`<span class="pill">${esc(x)}</span>`).join(' ')}</td><td>${r.max}</td></tr>`).join('');

  const topFamily=Object.entries(families).sort((a,b)=>b[1]-a[1])[0];
  const topPay=payEntries[0];
  const cityCounts=countBy(rows,d=>d.city);
  const topCity=Object.entries(cityCounts).sort((a,b)=>b[1]-a[1])[0];

  document.getElementById('payInsight').innerHTML=topPay
    ? `<b>${esc(topPay[0])}</b> lidera os pagamentos, com ${topPay[1]} vendas (${pct(topPay[1]/n)}).`
    : 'Sem dados para o filtro selecionado.';

  const insights=[];
  if(topFamily) insights.push(`<b>${esc(topFamily[0])}</b> é a família com maior presença no recorte, representando ${pct(topFamily[1]/n)} das vendas.`);
  if(leadYear) insights.push(`O <b>Model Year ${leadYear[0]}</b> concentra ${leadYear[1]} vendas, sendo o ano de modelo mais recorrente.`);
  if(topCity) insights.push(`<b>${esc(topCity[0])}</b> é a cidade com maior volume no recorte, com ${topCity[1]} venda(s).`);
  if(mf) insights.push(`Com o filtro de <b>${esc(mf)}</b>, a leitura passa a mostrar a distribuição regional e financeira desse modelo.`);
  document.getElementById('insights').innerHTML=insights.map(t=>`<div class="insight">${t}</div>`).join('') || '<div class="insight">Nenhum registro encontrado para os filtros selecionados.</div>';
}
function countBy(rows,fn){return rows.reduce((o,d)=>{const k=fn(d);o[k]=(o[k]||0)+1;return o},{})}
function drawEvolution(rows){
  const canvas=document.getElementById('salesEvolution');
  if(!canvas) return;
  const rect=canvas.getBoundingClientRect();
  const dpr=window.devicePixelRatio||1;
  const w=Math.max(rect.width,500), h=Math.max(rect.height,240);
  canvas.width=w*dpr; canvas.height=h*dpr;
  const ctx=canvas.getContext('2d'); ctx.scale(dpr,dpr);
  ctx.clearRect(0,0,w,h);

  const byMonth={};
  rows.forEach(d=>{
    const dt=new Date(d.date);
    if(Number.isNaN(dt.getTime())) return;
    const key=dt.getFullYear()+'-'+String(dt.getMonth()+1).padStart(2,'0');
    byMonth[key]=(byMonth[key]||0)+1;
  });
  const keys=Object.keys(byMonth).sort();
  const vals=keys.map(k=>byMonth[k]);
  if(!keys.length){
    ctx.fillStyle='#888'; ctx.font='12px Arial'; ctx.fillText('Sem dados para os filtros selecionados.',20,30); return;
  }

  const pad={l:45,r:20,t:18,b:42};
  const pw=w-pad.l-pad.r, ph=h-pad.t-pad.b;
  const max=Math.max(...vals,1);
  const niceMax=Math.max(4,Math.ceil(max/2)*2);

  ctx.strokeStyle='#e4e4e1'; ctx.lineWidth=1;
  ctx.font='10px Arial'; ctx.fillStyle='#777'; ctx.textAlign='right';
  for(let i=0;i<=4;i++){
    const y=pad.t+ph-(ph*i/4);
    ctx.beginPath(); ctx.moveTo(pad.l,y); ctx.lineTo(w-pad.r,y); ctx.stroke();
    ctx.fillText(String(Math.round(niceMax*i/4)),pad.l-8,y+3);
  }

  const points=keys.map((k,i)=>{
    const x=keys.length===1?pad.l+pw/2:pad.l+(i/(keys.length-1))*pw;
    const y=pad.t+ph-(vals[i]/niceMax)*ph;
    return [x,y];
  });

  // Area
  ctx.beginPath(); ctx.moveTo(points[0][0],pad.t+ph);
  points.forEach(p=>ctx.lineTo(p[0],p[1]));
  ctx.lineTo(points[points.length-1][0],pad.t+ph); ctx.closePath();
  ctx.fillStyle='rgba(213,0,28,.08)'; ctx.fill();

  // Line
  ctx.beginPath(); ctx.moveTo(points[0][0],points[0][1]);
  points.slice(1).forEach(p=>ctx.lineTo(p[0],p[1]));
  ctx.strokeStyle='#d5001c'; ctx.lineWidth=2.5; ctx.stroke();

  // Points
  points.forEach((p,i)=>{
    ctx.beginPath(); ctx.arc(p[0],p[1],4,0,Math.PI*2);
    ctx.fillStyle='#fff'; ctx.fill(); ctx.strokeStyle='#d5001c'; ctx.lineWidth=2; ctx.stroke();
  });

  // X labels, all periods preserved; rotate only when crowded.
  ctx.fillStyle='#777'; ctx.font='10px Arial'; ctx.textAlign='center';
  const step=keys.length>18?Math.ceil(keys.length/12):1;
  keys.forEach((k,i)=>{
    if(i%step!==0 && i!==keys.length-1) return;
    const [y,m]=k.split('-');
    const label=m+'/'+y.slice(2);
    ctx.fillText(label,points[i][0],h-17);
  });
}

function drawBars(id,entries,total){
  const max=Math.max(...entries.map(x=>x[1]),1);
  document.getElementById(id).innerHTML=entries.map(([k,v])=>`
    <div class="bar-row">
      <div class="bar-label" title="${esc(k)}">${esc(k)}</div>
      <div class="bar-track"><div class="bar-fill" style="width:${(v/max*100).toFixed(1)}%"></div></div>
      <div class="bar-num">${v}</div>
    </div>`).join('') || '<div style="color:#888;font-size:12px">Sem dados.</div>';
}
function drawDonut(entries,total){
  const donut=document.getElementById('donut');
  let start=0;
  const colors=entries.map((_,i)=>palette[i%palette.length]);
  const stops=entries.map(([k,v],i)=>{
    const a=start; start+=v/total*100;
    return `${colors[i]} ${a}% ${start}%`;
  });
  donut.style.background=entries.length?`conic-gradient(${stops.join(',')})`:'#ddd';
  document.getElementById('payLegend').innerHTML=entries.map(([k,v],i)=>`
    <div class="legend-item"><span class="dot" style="background:${colors[i]}"></span><span>${esc(k)}</span><strong style="margin-left:auto">${v}</strong></div>`).join('');
}
render();
